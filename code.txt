<script>
// Edit this list to match your real quizzes.
var QUIZZES=[
  {id:"gk",title:"General Knowledge Challenge",when:"Sat, 5 Dec 2026 · 6:00 PM IST"},
  {id:"sci",title:"Science & Tech Showdown",when:"Sat, 12 Dec 2026 · 6:00 PM IST"},
  {id:"cine",title:"Cinema & Pop Culture Night",when:"Sat, 19 Dec 2026 · 6:00 PM IST"}
];

var qEl=document.getElementById("quizzes"),
    form=document.getElementById("form"),
    msg=document.getElementById("msg"),
    go=document.getElementById("go");

QUIZZES.forEach(function(q,i){
  var l=document.createElement("label");l.className="quiz";
  var r=document.createElement("input");r.type="radio";r.name="quiz";r.value=q.id;if(i===0)r.checked=true;
  var d=document.createElement("div"),b=document.createElement("b"),s=document.createElement("span");
  b.textContent=q.title;s.textContent=q.when;
  d.appendChild(b);d.appendChild(s);l.appendChild(r);l.appendChild(d);qEl.appendChild(l);
});

function titleOf(id){for(var i=0;i<QUIZZES.length;i++)if(QUIZZES[i].id===id)return QUIZZES[i].title;return id}
function say(t,c){msg.textContent=t;msg.className="msg "+(c||"")}
function showDone(d){
  document.getElementById("doneText").textContent=d.name+", you're signed up for “"+titleOf(d.quiz)+"”. A confirmation will follow at "+d.email+".";
  document.getElementById("done").classList.remove("hide");
  form.classList.add("hide");
}

// Paste your form endpoint here (e.g. a Formspree, Getform or Google Apps Script URL).
var ENDPOINT="https://formspree.io/f/YOUR_FORM_ID";
var KEY="quiz-registered";

try{var prev=JSON.parse(localStorage.getItem(KEY)||"null");if(prev)showDone(prev)}catch(e){}

form.addEventListener("submit",async function(e){
  e.preventDefault();
  var name=document.getElementById("name").value.trim(),
      email=document.getElementById("email").value.trim(),
      team=document.getElementById("team").value.trim();
  var quiz=(form.querySelector("input[name=quiz]:checked")||{}).value;
  if(!name){say("Please enter your name.","err");return}
  if(!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)){say("Please enter a valid email.","err");return}
  go.disabled=true;say("Saving…");
  try{
    var d={name:name,email:email,team:team,quiz:quiz};
    var r=await fetch(ENDPOINT,{
      method:"POST",
      headers:{"Content-Type":"application/json","Accept":"application/json"},
      body:JSON.stringify(d)
    });
    if(!r.ok)throw new Error("bad status");
    try{localStorage.setItem(KEY,JSON.stringify(d))}catch(e){}
    showDone(d);
  }catch(err){
    say("Registration failed. Please try again.","err");
    go.disabled=false;
  }
});
